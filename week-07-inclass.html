1 Exercise 1: Medication String Cleanup (Cycle 1)

A clinic’s prescription log was hand-entered with inconsistent formatting.

prescriptions <- tibble(
  rx_id = c("R1", "R2", "R3", "R4", "R5", "R6"),
  drug  = c("Metformin 500mg BID", "LISINOPRIL 10mg", "metformin 850mg",
            "Atorvastatin 20mg", "METFORMIN 500mg bid", "amlodipine 5mg")
)
# 1. Add a column drug_lower with the drug text in lowercase (str_to_lower)
# 2. Filter to rows where drug_lower contains "metformin"
# 3. Add a dose column extracted with str_extract() and the pattern "\\d+mg"
# 4. How many prescriptions mention "bid" (any case)? Use str_detect() on
#    drug_lower and sum() the logical vector
metformin <- prescriptions |> mutate(drug_lower = str_to_lower(drug)) |>
  filter(str_detect(drug_lower, "metformin")) |>
  mutate(dose = str_extract(drug_lower, "\\d+mg")) 

sum(str_detect(metformin$drug_lower, "bid"))

3 Exercise 3: Ordered Regional Charts with forcats (Cycle 3)

# 1. From gm, keep 2008 rows, drop rows missing child_mortality or region,
#    group by region, and summarize mean_cm = mean(child_mortality)
# 2. Draw a horizontal bar chart: aes(x = mean_cm, y = region), geom_col().
#    Notice the unhelpful alphabetical ordering!
# 3. Redraw the chart with y = fct_reorder(region, mean_cm)
# 4. Separately: on the filtered 2008 rows, create region_top = fct_lump(region, n = 6)
#    and count(region_top, sort = TRUE)
filtered_gm <- gm |> filter(year == 2008) |>
  drop_na(child_mortality, region) |>
  group_by(region) |>
  summarize(mean_cm = mean(child_mortality))
ggplot(filtered_gm, aes(x = mean_cm, y = region)) +
  geom_col()
ggplot(filtered_gm, aes(x = mean_cm, y = fct_reorder(region, mean_cm))) +
  geom_col()
filtered_gm |> mutate(region_top = fct_lump(region, n = 6)) |>
  count(region_top, sort = TRUE)

4 Exercise 4: Dates & Length of Stay (Cycle 4 + AI Corner)

admissions <- tibble(
  patient   = c("A", "B", "C", "D"),
  dob       = c("1985-06-15", "1992-11-03", "1978-02-28", "2001-07-19"),
  admitted  = c("10/02/2026", "10/05/2026", "09/28/2026", "10/12/2026"),
  discharged = c("10/09/2026", "10/06/2026", "10/15/2026", "10/13/2026")
)
# 1. Parse dob with ymd(), and admitted/discharged with mdy() (mutate all three)
# 2. Add los_days: the length of stay in days, as.numeric(discharged - admitted)
# 3. Add age_years: floor(interval(dob, admitted) / years(1))
# 4. AI Corner: an AI computed age as (admitted - dob) / 365.25.
#    Compute that too (as.numeric(...) / 365.25) and compare: for which
#    patients does floor() of the two methods disagree? Why is interval() safer?
admissions |> mutate(dob = ymd(dob), admitted = mdy(admitted), discharged = mdy(discharged), los_days = as.numeric(discharged - admitted), age_years = floor(interval(dob, admitted) / years(1)), ai_age = as.numeric((admitted - dob) / 365.25))
