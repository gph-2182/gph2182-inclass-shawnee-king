sbp_clinic <- c(118, 152, 141, 129, 183, 136, 147)

2 Exercise 2: Outbreak Accumulator (Cycle 2)

daily_cases <- c(3, 7, 12, 20, 31, 26, 18, 40, 22, 15)
# 1. Write a for loop with an accumulator that builds cumulative_cases
#    (pre-allocate with numeric(length(daily_cases)), index with seq_along())
# 2. Verify your result equals cumsum(daily_cases) using identical() or all.equal()
# 3. Inside the SAME loop, maintain a second counter: how many days exceeded 25 cases?
# 4. Verify the counter equals sum(daily_cases > 25)
total_so_far <- numeric(length(daily_cases))
running_total <- 0
counter <- 0
for (i in seq_along(daily_cases)) {
  running_total <- running_total + daily_cases[i]
  total_so_far[i] <- running_total
  if (daily_cases[i] > 25){
    counter <- counter + 1
  }
}
all.equal(cumsum(daily_cases), total_so_far)
[1] TRUE
counter
[1] 3
sum(daily_cases > 25)
[1] 3

3 Exercise 3: Elimination Countdown with while (Cycle 3)

A vaccination campaign reduces monthly measles cases by 35%, starting from 5,000 cases.

# 1. Use a while loop to count how many months until cases fall below 50
# 2. Record the case count history in a vector as the loop runs
# 3. Add a guard rail so the loop can never run more than 100 months
# 4. Re-run with a 10% monthly reduction: how many months now?
cases <- 5000
months <- 0
case_count <- c()
while(cases >= 50 && months < 100) {
  cases <- cases * 0.65
  months <- months + 1
  case_count <- c(case_count, cases)
}
case_count

# 1. Write a for loop over continents <- c("Africa", "Asia", "Europe") that, for each,
#    filters gm to year 2008, that continent, non-missing life_expectancy >= 75,
#    and prints "<continent> has <n> countries with life expectancy >= 75"
# 2. AI Corner: an AI wrote this to flag high-mortality countries:
#      flags <- c()
#      for (i in 1:nrow(df)) { flags <- c(flags, if (df$child_mortality[i] > 50) "high" else "low") }
#    List two problems with it, then write the one-line vectorized replacement
#    using ifelse() on gm's child_mortality for 2008.
for (i in c("Africa", "Asia", "Europe")){
  filtered_gm <- gm |>
    filter(year == 2008, continent == i, !is.na(life_expectancy), life_expectancy >= 75) |>
    nrow()
  print(paste(i, "has", filtered_gm, "countries with life expectancy >= 75"))
}
