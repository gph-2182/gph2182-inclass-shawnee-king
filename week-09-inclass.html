# 1. Write incidence_rate(cases, population, per = 100000)
# 2. Use it inside mutate() on counties to add rate_100k; then add rate_1k with per = 1000
# 3. Add stopifnot() checks at the top of the function:
#    all populations are positive, and no cases value exceeds its population
# 4. Call incidence_rate(bad_counties$cases, bad_counties$population) and read the error.
#    Then replace the stopifnot() with if (...) stop("...") checks whose messages
#    tell a colleague exactly what is wrong
incidence_rate <- function(cases, population, per = 100000){
  if(!all(population > 0)){
    stop("Some population values are not positive.")
  }else if (!all(cases <= population)){
    stop("Some counties have more cases than population")
  }
  stopifnot(all(population > 0), all(cases <= population))
  cases / population * per
}
counties |>
  mutate(rate_100k = incidence_rate(cases, population), rate_1k = incidence_rate(cases, population, per = 1000))

summarize_continent <- function(data, ct, yr = 2008) {
  data |>
    filter(year == yr, continent == ct) |>
    summarize(
      continent = ct,
      n_countries = n(),
      median_cm = median(child_mortality, na.rm = TRUE)
    )
}
